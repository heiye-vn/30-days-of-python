## 一、文件基础操作

### 1.1 打开与关闭文件

Python 使用内置的 `open()` 函数打开文件，返回一个文件对象。推荐使用 `with` 语句（上下文管理器），它会在代码块结束后自动关闭文件，避免资源泄漏。

```python
# 推荐方式：使用 with 语句（自动关闭）
with open("data/sample.txt", "r", encoding="utf-8") as f:
    content = f.read()
    print(content)

# 不推荐：手动打开和关闭
f = open("data/sample.txt", "r", encoding="utf-8")
content = f.read()
f.close()  # 必须手动关闭，否则可能泄漏资源
```

### 1.2 读取文件

```python
# 读取全部内容
with open("data/sample.txt", "r", encoding="utf-8") as f:
    content = f.read()

# 按行读取（返回所有行的列表）
with open("data/sample.txt", "r", encoding="utf-8") as f:
    lines = f.readlines()
    for line in lines:
        print(line.strip())

# 逐行读取（内存友好，适合大文件）
with open("data/sample.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())

# 读取指定字节数
with open("data/sample.txt", "r", encoding="utf-8") as f:
    chunk = f.read(50)  # 读取前50个字符

# 读取一行
with open("data/sample.txt", "r", encoding="utf-8") as f:
    first_line = f.readline()
```

### 1.3 写入文件

```python
# 覆盖写入（"w" 模式会清空原内容）
with open("data/output.txt", "w", encoding="utf-8") as f:
    f.write("Hello, World!\n")
    f.write("Python file handling.\n")

# 写入多行
lines = ["第一行\n", "第二行\n", "第三行\n"]
with open("data/output.txt", "w", encoding="utf-8") as f:
    f.writelines(lines)

# 追加写入（"a" 模式不会清空原内容）
with open("data/output.txt", "a", encoding="utf-8") as f:
    f.write("追加的新内容\n")
```

### 1.4 文件模式一览

| 模式 | 说明 | 文件存在 | 文件不存在 |
|------|------|---------|-----------|
| `"r"` | 只读（默认） | 打开读取 | 报错 |
| `"w"` | 只写（覆盖） | 清空重写 | 创建新文件 |
| `"a"` | 追加写入 | 末尾追加 | 创建新文件 |
| `"x"` | 独占创建 | 报错 | 创建新文件 |
| `"r+"` | 读写 | 打开读写 | 报错 |
| `"w+"` | 读写（覆盖） | 清空重写 | 创建新文件 |
| `"a+"` | 读写（追加） | 打开读写 | 创建新文件 |
| `"b"` | 二进制模式 | 与其他模式组合 | 与其他模式组合 |
| `"t"` | 文本模式（默认） | 与其他模式组合 | 与其他模式组合 |

```python
# 二进制读取
with open("data/image.png", "rb") as f:
    binary_data = f.read()

# 二进制写入
with open("data/copy.png", "wb") as f:
    f.write(binary_data)
```

## 二、文件路径与目录操作

> [!TIP]
>
> 在底层的操作系统级别，所有的文件系统操作（如判断是否存在、读取、写入等）最终都必须定位到**绝对路径**。

### 2.1 

### 模块（推荐）

`pathlib` 是 Python 3.4+ 引入的现代路径操作库，提供面向对象的 API，是处理路径的首选方式。

```python
from pathlib import Path

# 创建路径对象
p = Path("data/sample.txt")
p_abs = Path("data/sample.txt").resolve()  # 绝对路径

# 路径拼接（使用 / 运算符）
base = Path("data")
file_path = base / "subdir" / "file.txt"
print(file_path)  # data\subdir\file.txt

# 获取路径信息
p = Path("data/sample.txt")
print(p.name)        # "sample.txt"（文件名）
print(p.stem)        # "sample"（不含扩展名）
print(p.suffix)      # ".txt"（扩展名）
print(p.parent)      # Path("data")（父目录）
print(p.parts)       # ("data", "sample.txt")
print(p.exists())    # True / False
print(p.is_file())   # True
print(p.is_dir())    # False

# 获取文件大小和修改时间
import os
p = Path("data/sample.txt")
stat = p.stat()
print(f"文件大小: {stat.st_size} bytes")
print(f"修改时间: {stat.st_mtime}")
```

### 2.2 os.path 模块

`os.path` 是传统的路径操作方式，仍然广泛使用。

```python
import os

# 路径拼接
path = os.path.join("data", "sample.txt")

# 路径拆分
dirname, filename = os.path.split("data/sample.txt")
name, ext = os.path.splitext("sample.txt")

# 路径信息
print(os.path.basename("data/sample.txt"))   # "sample.txt"
print(os.path.dirname("data/sample.txt"))    # "data"
print(os.path.abspath("data/sample.txt"))    # 绝对路径
print(os.path.exists("data/sample.txt"))     # True/False
print(os.path.isfile("data/sample.txt"))     # True/False
print(os.path.isdir("data"))                 # True/False
print(os.path.getsize("data/sample.txt"))    # 文件字节数
```

### 2.3 目录的增删改查

```python
import os
from pathlib import Path

# ========== 创建目录 ==========

# os 方式
os.mkdir("new_folder")             # 创建单层目录
os.makedirs("a/b/c", exist_ok=True)  # 递归创建多层目录

# pathlib 方式
Path("new_folder").mkdir(exist_ok=True)
Path("a/b/c").mkdir(parents=True, exist_ok=True)

# ========== 删除目录 ==========

# 删除空目录
os.rmdir("new_folder")
Path("new_folder").rmdir()

# 删除非空目录（需谨慎）
import shutil
shutil.rmtree("a")  # 递归删除整个目录树

# ========== 重命名/移动 ==========

os.rename("old_name", "new_name")          # 重命名
shutil.move("source_dir", "target_dir")    # 移动目录

# ========== 查询目录信息 ==========

# 列出目录内容
items = os.listdir("data")
for item in items:
    print(item)

# pathlib 遍历（更优雅）
data_dir = Path("data")
for item in data_dir.iterdir():
    print(f"{'[DIR]' if item.is_dir() else '[FILE]'} {item.name}")
```

### 2.4 文件遍历与匹配

```python
from pathlib import Path
import glob

# pathlib 匹配（推荐）
data_dir = Path("data")

# 匹配所有 .txt 文件
txt_files = list(data_dir.glob("*.txt"))
print(txt_files)

# 递归匹配所有 .txt 文件（包括子目录）
all_txt = list(data_dir.rglob("*.txt"))
print(all_txt)

# 匹配特定模式
py_files = list(Path(".").rglob("*.py"))

# glob 模块（传统方式）
files = glob.glob("data/*.txt")
files_recursive = glob.glob("data/**/*.txt", recursive=True)

# fnmatch 模块（文件名模式匹配）
import fnmatch
all_files = os.listdir("data")
txt_files = [f for f in all_files if fnmatch.fnmatch(f, "*.txt")]
```

## 三、文件与文件夹的高级操作

### 3.1 shutil 模块

`shutil` 提供高级文件操作：复制、移动、压缩等。

```python
import shutil
from pathlib import Path

# ========== 复制文件 ==========
shutil.copy("data/sample.txt", "data/sample_copy.txt")       # 复制文件内容+权限
shutil.copy2("data/sample.txt", "data/sample_copy2.txt")     # 复制文件+元数据（时间戳等）
shutil.copyfile("data/sample.txt", "data/sample_copy3.txt")  # 仅复制内容

# ========== 复制目录 ==========
shutil.copytree("data", "data_backup")  # 复制整个目录树

# ========== 移动文件/目录 ==========
shutil.move("data/sample_copy.txt", "data/moved/sample.txt")

# ========== 删除目录 ==========
shutil.rmtree("data_backup")  # 删除整个目录树

# ========== 磁盘使用情况 ==========
total, used, free = shutil.disk_usage("/")
print(f"Total: {total // (2**30)} GiB")
print(f"Used:  {used // (2**30)} GiB")
print(f"Free:  {free // (2**30)} GiB")

# ========== 获取文件类型 ==========
print(shutil.which("python"))  # 查找可执行文件路径
```

### 3.2 临时文件 tempfile

```python
import tempfile
from pathlib import Path

# 创建临时文件
with tempfile.NamedTemporaryFile(mode="w", suffix=".txt", delete=False) as f:
    f.write("临时数据")
    temp_path = f.name
    print(f"临时文件: {temp_path}")

# 手动删除
Path(temp_path).unlink()

# 创建临时目录
with tempfile.TemporaryDirectory() as tmpdir:

# SpooledTemporaryFile（内存+磁盘混合）
with tempfile.SpooledTemporaryFile(max_size=1024*1024) as f:

```

### 3.3 文件压缩与解压

```python
import zipfile
import tarfile
import shutil
from pathlib import Path

# ========== ZIP 压缩 ==========

# 创建 ZIP 文件
with zipfile.ZipFile("archive.zip", "w", zipfile.ZIP_DEFLATED) as zf:
    zf.write("data/sample.txt", "sample.txt")
    zf.write("data/sample.json", "sample.json")

# 解压 ZIP 文件
with zipfile.ZipFile("archive.zip", "r") as zf:
    zf.extractall("extracted/")         # 解压到目录
    file_list = zf.namelist()           # 查看文件列表
    content = zf.read("sample.txt")     # 读取文件内容

# 查看 ZIP 信息
with zipfile.ZipFile("archive.zip", "r") as zf:
    for info in zf.infolist():
        print(f"{info.filename}: {info.file_size} bytes")

# ========== TAR 压缩 ==========

# 创建 tar.gz 文件
with tarfile.open("archive.tar.gz", "w:gz") as tf:
    tf.add("data", arcname="data")

# 解压 tar.gz 文件
with tarfile.open("archive.tar.gz", "r:gz") as tf:
    tf.extractall("extracted_tar/")

# ========== shutil 快捷压缩 ==========

# 压缩整个目录
shutil.make_archive("backup", "zip", "data")        # → backup.zip
shutil.make_archive("backup", "gztar", "data")      # → backup.tar.gz

# 解压
shutil.unpack_archive("backup.zip", "output_dir")
```

from pathlib import Path

text = Path("data/sample.txt").read_text(encoding="utf-8")

### 4.2 CSV 文件

CSV（Comma-Separated Values）是最常见的结构化数据格式。

```python
import csv
from pathlib import Path

# ========== 标准库 csv ==========

# 读取 CSV
with open("data/sample.csv", "r", encoding="utf-8") as f:
    reader = csv.reader(f)
    header = next(reader)  # 读取表头
    print(f"列名: {header}")
    for row in reader:
        print(row)

# 以字典方式读取（推荐）
with open("data/sample.csv", "r", encoding="utf-8") as f:
    reader = csv.DictReader(f)
    for row in reader:
        print(f"{row['name']} - {row['city']} - {row['salary']}")

# 写入 CSV
data = [
    ["name", "age", "city"],
    ["Eve", 27, "Guangzhou"],
    ["Frank", 35, "Chengdu"],
]
with open("data/output.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.writer(f)
    writer.writerows(data)

# 以字典方式写入
fieldnames = ["name", "age", "city"]
rows = [
    {"name": "Eve", "age": 27, "city": "Guangzhou"},
    {"name": "Frank", "age": 35, "city": "Chengdu"},
]
with open("data/output.csv", "w", encoding="utf-8", newline="") as f:
    writer = csv.DictWriter(f, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerows(rows)

# ========== pandas（推荐处理大型 CSV） ==========
import pandas as pd

# 读取
df = pd.read_csv("data/sample.csv")
print(df.head())
print(df.describe())

# 写入
df.to_csv("data/output_pandas.csv", index=False, encoding="utf-8")

# 高级读取选项
df = pd.read_csv("data/sample.csv",
                  usecols=["name", "salary"],   # 只读特定列
                  dtype={"salary": int},           # 指定数据类型
                  nrows=10,                     # 只读前10行
                  skiprows=[1])                 # 跳过索引为 1 的行，索引 0 是表头，不能跳过
```

>[!TIP]
>
>在实际工程开发中，通常推荐**写操作**与**读操作**使用独立的 `with open` 代码块，分别使用 `w` 模式和 `r` 模式，这样可以有效避免文件指针管理不当带来的潜在 Bug，代码可读性也更好。

### 4.3 JSON 文件

JSON 是 Web 开发和数据交换中最常用的格式。

```python
import json
from pathlib import Path

# ========== 读取 JSON ==========

# 从文件读取
with open("data/sample.json", "r", encoding="utf-8") as f:
    data = json.load(f)
print(data["company"])
print(data["employees"][0]["name"])

# 从字符串读取
json_str = '{"name": "Alice", "age": 28}'
data = json.loads(json_str)

# ========== 写入 JSON ==========

data = {"name": "Bob", "skills": ["Python", "SQL", "Docker"]}

# 写入文件（格式化输出）
with open("data/output.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)

# 转为字符串
json_str = json.dumps(data, ensure_ascii=False, indent=2)

# ========== json.dumps/dump 常用参数 ==========
# ensure_ascii=False  允许输出非 ASCII 字符（中文等）
# indent=2            缩进空格数（美化输出）
# sort_keys=True      按 key 排序
# separators=(",", ":")  紧凑输出
# default=str         自定义不可序列化对象的处理函数

# ========== 处理特殊类型 ==========

from datetime import datetime

class CustomEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, datetime):
            return obj.isoformat()
        return super().default(obj)

data = {"timestamp": datetime.now(), "event": "login"}
json_str = json.dumps(data, cls=CustomEncoder, ensure_ascii=False)
```

### 4.4 XML 文件

## 八、大型文件的流式处理

当文件体积达到数百 MB 甚至数十 GB 时，`f.read()` 一次性加载到内存会导致内存溢出（OOM）。流式处理的核心思路是：**只读取当前需要处理的一小部分，处理完就释放，永远不把整个文件放进内存。**

### 8.1 流式处理核心思想

```
传统方式（一次性加载）：
  磁盘文件 ──全部读入──> 内存（可能 OOM）──> 处理

流式方式（惰性迭代）：
  磁盘文件 ──逐行/分块──> 缓冲区（固定大小）──> 处理 ──> 释放
             ↑ 循环 ↑
```

关键原则：

- **惰性求值**：用生成器（`yield`）代替列表，按需产出数据而不是预先加载。
- **固定内存**：无论文件多大，内存占用始终保持在可接受的范围内。
- **管道思维**：把处理流程想象成流水线，每个阶段只处理一小批数据然后传递给下一个阶段。

### 8.2 大文本文件流式读取

```python
from pathlib import Path

# ========== 方式一：直接迭代文件对象（最简洁） ==========
# 文件对象本身就是惰性迭代器，每次只读取一行
with open("huge_log.txt", "r", encoding="utf-8") as f:
    for line in f:
        # line 是当前行，处理完就会被 GC 回收
        if "ERROR" in line:
            print(line.strip())

# ========== 方式二：自定义分块生成器 ==========
def read_lines_batch(filepath, batch_size=1000):
    """每次产出一批行，适合需要批量处理的场景"""
    batch = []
    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            batch.append(line.strip())
            if len(batch) >= batch_size:
                yield batch
                batch = []
    if batch:  # 最后一批可能不满 batch_size
        yield batch

# 使用
for batch in read_lines_batch("huge_log.txt", batch_size=5000):
    print(f"处理了 {len(batch)} 行")
    # 批量插入数据库、写入新文件等

# ========== 方式三：按固定字节块读取 ==========
def read_chunks(filepath, chunk_size=64 * 1024):
    """按固定大小分块读取，适合不关心行边界的场景"""
    with open(filepath, "r", encoding="utf-8") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            yield chunk

for chunk in read_chunks("huge_log.txt"):
    # 统计关键词出现次数等
    count = chunk.count("ERROR")

# ========== 方式四：带进度条的流式读取 ==========
# pip install tqdm
from tqdm import tqdm

# 先获取文件总大小，用于显示进度
file_size = Path("huge_log.txt").stat().st_size

with open("huge_log.txt", "r", encoding="utf-8") as f:
    with tqdm(total=file_size, unit="B", unit_scale=True) as pbar:
        for line in f:
            process(line)
            pbar.update(len(line.encode("utf-8")))
```

with open("big_data.json", "rb") as f:

### 8.5 大 XML 流式解析

标准 `ET.parse()` 会将整个 XML 树加载到内存。`iterparse()` 可以边解析边处理，处理过的节点会被丢弃。

```python
import xml.etree.ElementTree as ET

# ========== iterparse：边解析边处理 ==========
# 假设 big_data.xml 有数百万个 <record> 节点

count = 0
for event, elem in ET.iterparse("big_data.xml", events=("end",)):
    if elem.tag == "record":
        # 处理当前 record 节点
        name = elem.find("name").text
        value = elem.find("value").text
        count += 1

        # ⚠️ 关键：处理完后清除节点，释放内存
        elem.clear()

print(f"共处理 {count} 条记录")

# ========== 带父节点清除的完整方案 ==========
# 对于特别大的文件，还需要清除父节点的引用
context = ET.iterparse("big_data.xml", events=("start", "end"))
context = iter(context)

# 获取根元素
event, root = next(context)

for event, elem in context:
    if event == "end" and elem.tag == "record":
        # 处理数据
        process_record(elem)

        # 清除已处理的节点
        root.clear()

# ========== lxml 的 iterparse（更快） ==========
# pip install lxml
from lxml import etree

context = etree.iterparse("big_data.xml", tag="record")
for event, elem in context:
    process_record(elem)
    elem.clear()
    # 清除父节点中对已处理子节点的引用
    while elem.getprevious() is not None:
        del elem.getparent()[0]
```

### 8.6 二进制大文件分块处理

适用于大文件复制、哈希计算、加密、压缩等场景。

```python
import hashlib
import shutil
from pathlib import Path

# ========== 大文件 MD5/SHA256 计算 ==========
def file_hash(filepath, algorithm="sha256", chunk_size=64 * 1024):
    """流式计算文件哈希，不将整个文件读入内存"""
    h = hashlib.new(algorithm)
    with open(filepath, "rb") as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            h.update(chunk)
    return h.hexdigest()

print(file_hash("big_file.zip"))

# ========== 大文件流式复制（带进度） ==========
from tqdm import tqdm

def stream_copy(src, dst, chunk_size=1024 * 1024):
    """流式复制大文件，内存占用仅 chunk_size"""
    total = Path(src).stat().st_size
    with open(src, "rb") as fin, open(dst, "wb") as fout:
        with tqdm(total=total, unit="B", unit_scale=True, desc="复制中") as pbar:
            while True:
                chunk = fin.read(chunk_size)
                if not chunk:
                    break
                fout.write(chunk)
                pbar.update(len(chunk))

stream_copy("big_video.mp4", "backup/big_video.mp4")

# ========== 大文件流式加密 ==========
from hashlib import pbkdf2_hmac
# 实际项目推荐用 cryptography 库

# ========== 大文件对比（分块比较） ==========
def files_equal(file1, file2, chunk_size=64 * 1024):
    """流式对比两个文件是否相同"""
    if Path(file1).stat().st_size != Path(file2).stat().st_size:
        return False
    with open(file1, "rb") as f1, open(file2, "rb") as f2:
        while True:
            chunk1 = f1.read(chunk_size)
            chunk2 = f2.read(chunk_size)
            if chunk1 != chunk2:
                return False
            if not chunk1:
                return True
```

### 8.7 mmap 内存映射文件

`mmap` 将文件映射到虚拟内存，可以像操作内存一样操作文件内容，由操作系统管理磁盘和内存之间的数据交换。特别适合需要随机访问大文件某个位置的场景。

```python
import mmap

# ========== 基本用法 ==========
with open("huge_data.bin", "rb") as f:
    # 将整个文件映射到内存（0 表示映射整个文件）
    with mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ) as mm:
        # 像操作 bytes 一样操作文件
        print(mm[0:10])         # 读取前 10 字节
        print(mm[-10:])         # 读取最后 10 字节

        # 查找
        pos = mm.find(b"TARGET")
        if pos != -1:
            print(f"在偏移量 {pos} 处找到")

        # 遍历
        offset = 0
        while True:
            pos = mm.find(b"ERROR", offset)
            if pos == -1:
                break
            print(f"找到 ERROR 在位置 {pos}")
            offset = pos + 1

# ========== 可写映射 ==========
with open("data.bin", "r+b") as f:
    mm = mmap.mmap(f.fileno(), 0)
    mm[0:5] = b"Hello"    # 直接修改文件内容
    mm.flush()             # 确保写入磁盘
    mm.close()

# ========== 搜索大文件中的模式（结合正则） ==========
import re

with open("huge_log.txt", "rb") as f:
    with mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ) as mm:
        # 在映射上执行正则搜索
        pattern = re.compile(rb"ERROR.*?(\d{4}-\d{2}-\d{2})")
        for match in pattern.finditer(mm):
            print(f"日期: {match.group(1).decode()}")

# 适用场景 vs 不适用场景：
# ✅ 适合：随机访问、搜索/替换、固定格式二进制解析
# ❌ 不适合：纯顺序读取（直接 for line in f 更快）
```

### 8.8 流式写入与管道处理

```python
import gzip
import bz2

# ========== 边读边写：流式过滤 ==========
def filter_log(input_path, output_path, keyword="ERROR"):
    """从大日志中过滤出包含关键词的行，写入新文件"""
    with open(input_path, "r", encoding="utf-8") as fin, \
         open(output_path, "w", encoding="utf-8") as fout:
        for line in fin:
            if keyword in line:
                fout.write(line)

filter_log("huge_access.log", "errors_only.log")

# ========== 流式压缩写入 ==========
with gzip.open("output.txt.gz", "wt", encoding="utf-8") as f:
    for chunk in generate_data():
        f.write(chunk)  # 边生成边压缩边写入

# 流式读取压缩文件
with gzip.open("output.txt.gz", "rt", encoding="utf-8") as f:
    for line in f:
        process(line)

# bz2 压缩（压缩率更高，速度稍慢）
with bz2.open("output.txt.bz2", "wt", encoding="utf-8") as f:
    for chunk in generate_data():
        f.write(chunk)

# ========== 管道式处理链 ==========
def pipeline(source_file, *filters):
    """构建处理管道：source -> filter1 -> filter2 -> ..."""
    def read_lines(path):
        with open(path, "r", encoding="utf-8") as f:
            yield from f

    stream = read_lines(source_file)
    for f in filters:
        stream = f(stream)
    yield from stream

# 定义过滤器
def remove_empty(lines):
    return (line for line in lines if line.strip())

def to_upper(lines):
    return (line.upper() for line in lines)

def filter_errors(lines):
    return (line for line in lines if "ERROR" in line)

# 组装管道
for line in pipeline("app.log", remove_empty, to_upper, filter_errors):
    print(line.strip())
```

### 8.9 并发加速流式处理

当处理逻辑比较耗时（如网络请求、复杂计算）时，可以用多进程/多线程并行处理流式数据。

```python
from concurrent.futures import ProcessPoolExecutor, ThreadPoolExecutor
import multiprocessing as mp

# ========== 多进程并行处理（CPU 密集型） ==========
def process_chunk(chunk):
    """处理一批数据（在子进程中运行）"""
    results = []
    for line in chunk:
        # 耗时计算
        result = heavy_computation(line)
        results.append(result)
    return results

def parallel_process(filepath, num_workers=4, chunk_size=10000):
    """多进程并行处理大文件"""
    with open(filepath, "r", encoding="utf-8") as f:
        # 跳过表头
        header = next(f)

        batch = []
        all_results = []

        with ProcessPoolExecutor(max_workers=num_workers) as executor:
            futures = []
            for line in f:
                batch.append(line.strip())
                if len(batch) >= chunk_size:
                    futures.append(executor.submit(process_chunk, batch))
                    batch = []

            if batch:
                futures.append(executor.submit(process_chunk, batch))

            for future in futures:
                all_results.extend(future.result())

    return all_results

# ========== 多线程并行处理（IO 密集型） ==========
def fetch_url(url):
    """网络请求（在子线程中运行）"""
    import urllib.request
    return urllib.request.urlopen(url).read()

def parallel_fetch(url_file, max_workers=10):
    """从文件中读取 URL 列表，并发下载"""
    with open(url_file, "r", encoding="utf-8") as f:
        urls = [line.strip() for line in f if line.strip()]

    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        results = list(executor.map(fetch_url, urls))

    return results

# ========== multiprocessing.Pool + imap（更简洁） ==========
def process_line(line):
    return line.strip().upper()

with mp.Pool(processes=4) as pool:
    with open("huge_file.txt", "r", encoding="utf-8") as f:
        # imap 惰性迭代，不会一次性加载所有数据
        for result in pool.imap(process_line, f, chunksize=1000):
            print(result)
```

### 8.10 实战：大日志文件分析

综合应用上述技巧，对一个假设的大型日志文件（10GB+）做分析。

```python
import re
from collections import Counter, defaultdict
from pathlib import Path
from tqdm import tqdm

LOG_PATTERN = re.compile(
    r"(?P<timestamp>\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2})\s+"
    r"(?P<level>\w+)\s+"
    r"(?P<module>[\w.]+)\s+"
    r"(?P<message>.+)"
)

def analyze_log(filepath):
    """流式分析大型日志文件"""
    level_counts = Counter()
    hourly_counts = defaultdict(int)
    error_messages = []

    file_size = Path(filepath).stat().st_size

    with open(filepath, "r", encoding="utf-8") as f:
        with tqdm(total=file_size, unit="B", unit_scale=True) as pbar:
            for line in f:
                pbar.update(len(line.encode("utf-8")))

                match = LOG_PATTERN.match(line)
                if not match:
                    continue

                level = match.group("level")
                timestamp = match.group("timestamp")
                message = match.group("message")

                # 统计各级别数量
                level_counts[level] += 1

                # 统计每小时错误数
                if level == "ERROR":
                    hour = timestamp[:13]  # "2026-07-01 14"
                    hourly_counts[hour] += 1

                    # 只保留最近的 100 条错误（控制内存）
                    error_messages.append(message)
                    if len(error_messages) > 100:
                        error_messages.pop(0)

    # 输出报告
    print("=== 日志分析报告 ===")
    print(f"\n日志级别分布:")
    for level, count in level_counts.most_common():
        print(f"  {level}: {count:,}")

    print(f"\n每小时错误数 (Top 10):")
    for hour, count in sorted(hourly_counts.items(), key=lambda x: -x[1])[:10]:
        print(f"  {hour}:00 - {count} errors")

    print(f"\n最近的错误信息:")
    for msg in error_messages[-10:]:
        print(f"  - {msg[:100]}")

# 运行分析
analyze_log("server_access.log")
```

### 8.11 流式处理方案选型速查

| 场景 | 推荐方案 | 关键代码/库 |
|------|---------|------------|
| 大文本逐行处理 | 直接迭代文件对象 | `for line in f:` |
| 大文本批量处理 | 分批生成器 | `yield batch` |
| 大 CSV 数据分析 | pandas chunksize | `pd.read_csv(f, chunksize=N)` |
| 大 CSV 逐行处理 | csv.DictReader | `for row in reader:` |
| 大 JSON 数组 | ijson 流式解析 | `ijson.items(f, "arr.item")` |
| JSON 数据交换 | JSON Lines 格式 | 每行一个 `json.dumps()` |
| 大 XML 节点遍历 | ET.iterparse | `iterparse(f, events=("end",))` |
| 大文件哈希/对比 | 分块读取 | `f.read(chunk_size)` 循环 |
| 大文件复制 | 分块读写 + tqdm | 自定义 `stream_copy()` |
| 大文件随机访问 | mmap 内存映射 | `mmap.mmap(f.fileno(), 0)` |
| 大文件压缩读写 | gzip/bz2.open | `gzip.open(f, "rt")` |
| CPU 密集型并行 | ProcessPoolExecutor | `executor.submit()` + 分块 |
| IO 密集型并行 | ThreadPoolExecutor | `executor.map()` |
| 流式管道处理 | 生成器链 | `yield from` 串联 |
